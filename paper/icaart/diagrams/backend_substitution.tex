%% ICAART: one figure for the whole system and the substitution design.
%% Merged from the former backend_substitution + pipeline_end_to_end drawings,
%% which duplicated the same spine. The per-agent execution order is no longer
%% drawn as separate boxes: the three VLM agents are collapsed into the single
%% block that the backend swap replaces, since the paper's claim is about that
%% block as a unit. Everything above the fork is computed once and reused
%% bit-identically by both runs; the triage rule below the fork is one revision
%% applied to both.
%%
%% Type sizes: the figure is \resizebox'd to \columnwidth, so on-page text size
%% is (font size) x (\columnwidth / natural width). Labels are one step larger
%% than they used to be (\small / \scriptsize instead of \scriptsize / \tiny),
%% and the long single-line captions were reflowed so the natural width stays
%% near 10cm; otherwise the wider drawing would scale the larger type straight
%% back down.
\begin{tikzpicture}[

                    minimum width=3.8cm, minimum height=0.75cm},
  swinblock/.style={draw, rounded corners=2pt, align=center, fill=orange!20,
                    minimum width=3.8cm, minimum height=0.75cm},
  detblock/.style ={draw, rounded corners=2pt, align=center, fill=gray!14,
                    minimum width=3.15cm, minimum height=1.15cm},
