% A candidate is formed by subtracting SAM's coverage from the Swin class map,
% so on an object the pipeline already had, all that is left is the rim around
% it -- drawn here as the annulus it literally is, not as a blob alongside.
\newcommand{\rimglyph}[2]{%
  \begin{scope}[shift={(#1)}]
    \draw[ring] (0,0) circle (0.78);
    \fill[#2, even odd rule] (0,0) ellipse (0.56 and 0.44)
                             (0,0) ellipse (0.42 and 0.30);
    \fill[sam] (0,0) ellipse (0.42 and 0.30);
  \end{scope}}
\newcommand{\aloneglyph}[2]{%
  \begin{scope}[shift={(#1)}]
    \draw[ring] (0,0) circle (0.78);
    \fill[#2] (0,0) ellipse (0.34 and 0.27);
  \end{scope}}
